async function submitAuth() { if (!state.dataStore) { showToast('目前處於離線/未設定模式，無法使用登入功能', 'danger'); return; } const isLogin = state.authMode === 'login'; const email = emailInput.value.trim(); const password = passwordInput.value; const displayName = displayNameInput.value.trim(); if (!email || !password || (!isLogin && !displayName)) { showToast('請填寫所有必填欄位', 'danger'); return; } setLoading(true); try { if (isLogin) { const user = await state.dataStore.login({ email, password }); state.currentUser = user; showToast(`登入成功！歡迎 ${user.displayName}`); } else { // ── 修正重點：改為物件形式傳入，對齊解構賦值 ── const user = await state.dataStore.register({ displayName, email, password }); state.currentUser = user; showToast('註冊成功！已為您自動登入'); } closeAuthModal(); setStep('main'); await syncCollectionFromCloud(); } catch (error) { console.error('認證失敗:', error); showToast(error.message || '認證失敗，請檢查輸入資料', 'danger'); } finally { setLoading(false); } }
